# Платформа приложения: то, что сервисам и приложению нужно от системы, —
# мьютекс, очередь, время, задача, журнал (platform.h).
#
# Компонент самостоятельный и в стороне от ядра API: ядро (api/) платформы НЕ
# знает и её заголовков не включает. Всё, что ядру нужно от системы, приложение
# передаёт ему таблицей api_system_t (см. api/include/api.h), собранной НА ЭТОЙ
# платформе (main/app.c). Обратной зависимости нет: platform не зависит ни от
# api, ни от transport — только от FreeRTOS/ESP-IDF.
#
# Задачи и очереди здесь остаются намеренно: cooperative-контракт ядра говорит,
# что ядро их не заводит, но сервис вправе держать свой поток и свою очередь как
# ВНУТРЕННЕЕ дело (он же и отвечает за их синхронизацию). Ядру они не видны.
idf_component_register(
    SRCS "src/platform_freertos.c"
    INCLUDE_DIRS "include"
)
