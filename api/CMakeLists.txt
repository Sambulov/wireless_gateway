# Ядро API: вызовы, подписки, регистрация сервисов.
#
# Кооперативный контракт (главное требование к этому компоненту):
#   * ядро НЕ заводит ни потоков, ни очередей — такт ему даёт вызывающий
#     (api_worker), а ответ клиенту отправляет сам обработчик из своего вызова
#     (bApiCallSendJson / bApiCallSendJsonFidGroup);
#   * ядро НЕ спит и ничего не ждёт: единственная синхронизация — мьютекс, а
#     «подождать» значит вернуть управление (такт повторится);
#   * всё, что ядру нужно от системы, — мьютекс, чтение таймера и журнал, и
#     каждый такой вызов принимает системный контекст (api_system_t в api.h).
#
# Отсюда — зависимости этого компонента:
#   * platform НЕ подключён: системные вызовы приходят таблицей, собранной
#     приложением (main/app.c). Заголовков platform.h в компоненте нет вообще —
#     ни одного системного имени, это проверяется отдельной проверкой в отчёте
#     (поиск по каталогу api/ не находит ни одного);
#   * транспорт — отдельный компонент (transport.h): ядро подписывается на его
#     события через delegate_t и отдаёт сообщения через begin_send, но сам
#     контракт живёт не здесь.
idf_component_register(
    SRCS "src/api_core.c"
    INCLUDE_DIRS "include"
    REQUIRES CodeLibrary json transport
)
