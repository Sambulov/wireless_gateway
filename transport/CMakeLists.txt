# Транспорт — самостоятельный компонент: контракт (transport.h) и ничего
# больше. Исходников у него нет — реализация живёт у того, кто транспорт
# поднимает (в прошивке main/ws_transport.c), и подключается к контракту, а не
# наоборот. Поэтому компонент — только каталог заголовков.
#
# Зависимость ровно одна и не на ядро API: CodeLibrary нужен ради delegate_t и
# event_t — того же механизма подписки, что у UART и линии связи. Про ядро API
# транспорт не знает ничего: подписчик для него — делегат, а кто он, ему
# безразлично.
idf_component_register(
    INCLUDE_DIRS "include"
    REQUIRES CodeLibrary
)
